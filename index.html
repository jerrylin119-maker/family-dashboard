// ========================================================================= // 1. 初始化工作表 (第一次貼上程式碼後，請先在上方選單選擇此函數並按「執行」) // ========================================================================= function initSheets() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheets = ['Timmy', 'Chloe', 'Night_Timmy', 'Night_Chloe', 'Study_Timmy', 'Notes']; // 建立任務清單工作表 sheets.forEach(name => { let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.appendRow(['ID', 'Text', 'Done']); } }); // 建立刷牙打卡工作表 let brushSheet = ss.getSheetByName('Brush'); if (!brushSheet) { brushSheet = ss.insertSheet('Brush'); brushSheet.appendRow(['Member', 'Morning', 'Night']); brushSheet.appendRow(['爸爸', false, false]); brushSheet.appendRow(['媽媽', false, false]); brushSheet.appendRow(['Timmy', false, false]); brushSheet.appendRow(['Chloe', false, false]); } // 建立家事工作表 let choreSheet = ss.getSheetByName('Chores'); if (!choreSheet) { choreSheet = ss.insertSheet('Chores'); choreSheet.appendRow(['Day', 'Person', 'Chore', 'TodayStatus', 'TodayDone', 'DoneTime']); choreSheet.appendRow(['週日', '爸爸', '大掃除清貓砂', '今日打卡狀態 ➡️', false, '']); choreSheet.appendRow(['週一', 'TIMMY', '清貓砂', '', '', '']); choreSheet.appendRow(['週二', 'CHLOE', '清貓砂(含白桶)', '', '', '']); choreSheet.appendRow(['週三', 'TIMMY', '清貓砂', '', '', '']); choreSheet.appendRow(['週四', 'CHLOE', '清貓砂', '', '', '']); choreSheet.appendRow(['週五', 'TIMMY', '清貓砂(含白桶)', '', '', '']); choreSheet.appendRow(['週六', 'CHLOE', '清貓砂', '', '', '']); } } // ========================================================================= // 2. 每日午夜自動重置功能 (請綁定觸發條件：時間驅動 -> 日計時器 -> 午夜至上午1點) // ========================================================================= function autoDailyReset() { const ss = SpreadsheetApp.getActiveSpreadsheet(); // 清空所有任務的打勾狀態 (將第3欄 Done 設為 false) const taskSheets = ['Timmy', 'Chloe', 'Night_Timmy', 'Night_Chloe', 'Study_Timmy']; taskSheets.forEach(name => { let sheet = ss.getSheetByName(name); if (sheet) { let lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.getRange(2, 3, lastRow - 1, 1).setValue(false); } } }); // 爸媽叮嚀通常不強制清除打勾，若要清除可把 'Notes' 加入上方陣列 // 清空刷牙紀錄 (將 B 欄與 C 欄設為 false) let brushSheet = ss.getSheetByName('Brush'); if (brushSheet) { let lastRow = brushSheet.getLastRow(); if (lastRow > 1) { brushSheet.getRange(2, 2, lastRow - 1, 2).setValue(false); } } // 清空今日家事打卡 (將 E2 設為 false，F2 設為空字串) let choreSheet = ss.getSheetByName('Chores'); if (choreSheet) { choreSheet.getRange("E2").setValue(false); choreSheet.getRange("F2").setValue(""); } } // ========================================================================= // 3. 處理 GET 請求 (提供看板讀取資料) // ========================================================================= function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let result = {}; // 讀取清單資料 const sheetMap = { 'morning_timmy': 'Timmy', 'morning_chloe': 'Chloe', 'night_timmy': 'Night_Timmy', 'night_chloe': 'Night_Chloe', 'study_timmy': 'Study_Timmy', 'parent_notes': 'Notes' }; for (let key in sheetMap) { let sheet = ss.getSheetByName(sheetMap[key]); result[key] = []; if (sheet && sheet.getLastRow() > 1) { let data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 3).getValues(); data.forEach(row => { if (row[0] && row[1]) { result[key].push({ id: row[0], text: row[1], done: !!row[2] }); } }); } } // 讀取刷牙紀錄 result.brush = {}; let brushSheet = ss.getSheetByName('Brush'); if (brushSheet && brushSheet.getLastRow() > 1) { let bData = brushSheet.getRange(2, 1, brushSheet.getLastRow() - 1, 3).getValues(); bData.forEach(row => { if (row[0]) result.brush[row[0]] = { morning: !!row[1], night: !!row[2] }; }); } // 讀取家事紀錄 result.chores = []; result.chore_today_done = { done: false, time: '' }; let choreSheet = ss.getSheetByName('Chores'); if (choreSheet && choreSheet.getLastRow() > 1) { let cData = choreSheet.getRange(2, 1, choreSheet.getLastRow() - 1, 3).getValues(); cData.forEach(row => { if (row[0]) result.chores.push({ day: row[0], person: row[1], chore: row[2] }); }); // E2: 狀態, F2: 時間 result.chore_today_done = { done: !!choreSheet.getRange("E2").getValue(), time: choreSheet.getRange("F2").getValue().toString() // 強制轉字串避免日期物件 }; } const callback = e && e.parameter && e.parameter.callback; if (callback) { return ContentService.createTextOutput(callback + '(' + JSON.stringify(result) + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // ========================================================================= // 4. 處理 POST 請求 (接收看板的寫入動作) // ========================================================================= function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const payload = JSON.parse(e.postData.contents); const action = payload.action; // 新增任務 if (action === 'add_task') { let sheet = ss.getSheetByName(payload.sheetName); if (sheet) sheet.appendRow([Date.now(), payload.text, false]); } // 切換任務打勾狀態 if (action === 'toggle_task') { let sheet = ss.getSheetByName(payload.sheetName); if (sheet) { let data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] == payload.rowId) { sheet.getRange(i + 1, 3).setValue(payload.done); break; } } } } // 刷牙打卡 if (action === 'brush_check') { let sheet = ss.getSheetByName('Brush'); if (sheet) { let data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === payload.member) { let col = payload.session === 'morning' ? 2 : 3; sheet.getRange(i + 1, col).setValue(true); break; } } } } // 今日家事完成打卡 (完美解決 1899 年 Bug) if (action === 'chore_check') { let sheet = ss.getSheetByName('Chores'); if (sheet) { sheet.getRange("E2").setValue(true); // 使用 Utilities.formatDate 強制寫入純文字格式的時間，避免 Google 試算表亂轉 1899 年 let timeStr = Utilities.formatDate(new Date(), "GMT+8", "HH:mm"); sheet.getRange("F2").setValue(`'${timeStr}`); // 前方加單引號強制為文字格式 } } // 更新整週家事表 if (action === 'update_chores') { let sheet = ss.getSheetByName('Chores'); if (sheet && payload.chores) { for (let i = 0; i < payload.chores.length; i++) { let c = payload.chores[i]; sheet.getRange(i + 2, 1, 1, 3).setValues([[c.day, c.person, c.chore]]); } } } // 接收前端跨日強制重置的請求 if (action === 'daily_reset') { autoDailyReset(); } return ContentService.createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
